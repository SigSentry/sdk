SigSentry Embed Demo

Vanilla JS integration — no React required in your app

Inline Widget

<div id="widget"></div>
<script>
  SigSentry.init({
    target: '#widget',
    apiKey: 'ss_live_...',
    baseUrl: 'http://localhost:3001',
    theme: 'dark',
    mode: 'inline',
  });
</script>
